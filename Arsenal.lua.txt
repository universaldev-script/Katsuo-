local function isAlive(char)
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and hum.Health > 0 then return true end
    return false

local function isEnemy(player)
    if ESPState.TeamCheck == "Everyone" then return player ~= LocalPlayer end
    if ESPState.TeamCheck == "Allies" then return player.Team == LocalPlayer.Team end
    return player.Team ~= LocalPlayer.Team
end
local function worldToScreen(pos)
    local cam = Workspace.CurrentCamera
    if not cam then return Vector2.new(0,0), false, 0 end
    local vec, onScreen = cam:WorldToViewportPoint(pos)
    return Vector2.new(vec.X, vec.Y), onScreen, vec.Z
end
local function getDistance(char)
    if not char then return math.huge end
    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso") or char.PrimaryPart
    if not hrp then return math.huge end
    local cam = Workspace.CurrentCamera
    return (hrp.Position - cam.CFrame.Position).Magnitude
end
local function isVisible(part)
    local origin = Workspace.CurrentCamera.CFrame.Position
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.IgnoreWater = true
    local filter = {Workspace.CurrentCamera}
    local lc = getLocalCharacter()
    if lc then table.insert(filter, lc) end
    params.FilterDescendantsInstances = filter
    local cf = part.CFrame
    local size = part.Size
    local offsets = {
        Vector3.new(0,0,0),
        Vector3.new(size.X/2.2,size.Y/2.2,0),
        Vector3.new(-size.X/2.2,size.Y/2.2,0),
        Vector3.new(size.X/2.2,-size.Y/2.2,0),
        Vector3.new(-size.X/2.2,-size.Y/2.2,0),
    }
    local hits = 0
    for _,off in ipairs(offsets) do
        local target = cf * off
        local dir = target - origin
        local res = Workspace:Raycast(origin, dir + dir.Unit, params)
        if res then
            if res.Instance:IsDescendantOf(part.Parent) then hits = hits + 1 end
        else
            hits = hits + 1
        end
    end
    return hits >= 3
end

local function updateESP(char)
    local obj=ESP_OBJECTS[char]
    if not obj then return end
    local hum=char:FindFirstChildOfClass("Humanoid")
    local hrp=char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso") or char.PrimaryPart
    local active=ESPState.Enabled and hrp and isAlive(char)
    if not active then hideAll(obj) return end
    local player=ESP_PLAYER_MAP[char] or PlayerService:GetPlayerFromCharacter(char)
    if player and not isEnemy(player) then hideAll(obj) return end
    local hrpPos,onScreen,depth=worldToScreen(hrp.Position)
    local dist=getDistance(char)
    if not onScreen or depth<=0 or dist>ESPState.MaxDistance then hideAll(obj) return end
    local topScreen=worldToScreen(hrp.Position+Vector3.new(0,3.1,0))
    local footScreen=worldToScreen(hrp.Position-Vector3.new(0,3.0,0))
    local height=math.max(math.abs(footScreen.Y-topScreen.Y),2)
    local width=height*0.55
    local vis=isVisible(hrp)
    local dynamicClr=vis and ESPState.VisibleBoxColor or ESPState.BoxColor
    local tracerClr=ESPState.TracerColor
    local nameClr=ESPState.NameColor
    local distClr=ESPState.DistanceColor
    local headDotClr=ESPState.HeadDotColor
    local skelClr=vis and ESPState.VisibleBoxColor or ESPState.SkeletonColor
    local centerX=hrpPos.X
    local left=centerX-width/2
    local right=centerX+width/2
    local top=topScreen.Y
    local bot=footScreen.Y
    local style=ESPState.BoxStyle
    local showFull=(style=="Full Box") and ESPState.Box
    local showCorner=(style=="Corner Box") and ESPState.Box
    obj.Box.Visible=showFull
    obj.BoxOutline.Visible=showFull
    if showFull then
        obj.Box.Size=Vector2.new(width,height)
        obj.Box.Position=Vector2.new(left,top)
        obj.Box.Color=dynamicClr
        obj.Box.Thickness=ESPState.BoxThickness
        obj.BoxOutline.Size=Vector2.new(width,height)
        obj.BoxOutline.Position=Vector2.new(left,top)
        obj.BoxOutline.Thickness=ESPState.BoxThickness+2
    end
    obj.BoxFill.Visible=showFull and ESPState.BoxFill
    if obj.BoxFill.Visible then
        obj.BoxFill.Size=Vector2.new(width,height)
        obj.BoxFill.Position=Vector2.new(left,top)
        obj.BoxFill.Color=dynamicClr
        obj.BoxFill.Transparency=ESPState.FillTransparency
    end
    local cx,cy=width*0.25,height*0.25
    local corners={
        {Vector2.new(left,top),Vector2.new(left+cx,top),Vector2.new(left,top+cy)},
        {Vector2.new(right,top),Vector2.new(right-cx,top),Vector2.new(right,top+cy)},
        {Vector2.new(left,bot),Vector2.new(left+cx,bot),Vector2.new(left,bot-cy)},
        {Vector2.new(right,bot),Vector2.new(right-cx,bot),Vector2.new(right,bot-cy)},
    }
    local li=1
    for _,c in ipairs(corners) do
        obj.CornerLines[li].From=c[1] obj.CornerLines[li].To=c[2] obj.CornerLines[li].Color=dynamicClr obj.CornerLines[li].Visible=showCorner obj.CornerLines[li].Thickness=ESPState.BoxThickness
        obj.CornerLines[li+1].From=c[1] obj.CornerLines[li+1].To=c[3] obj.CornerLines[li+1].Color=dynamicClr obj.CornerLines[li+1].Visible=showCorner obj.CornerLines[li+1].Thickness=ESPState.BoxThickness
        li = li + 2
    end
    obj.Tracer.Visible=ESPState.Tracer
    if ESPState.Tracer then
        local cam=Workspace.CurrentCamera
        local origin=Vector2.new(cam.ViewportSize.X/2,cam.ViewportSize.Y)
        if ESPState.TracerOrigin=="Top" then origin=Vector2.new(cam.ViewportSize.X/2,0)
        elseif ESPState.TracerOrigin=="Center" then origin=Vector2.new(cam.ViewportSize.X/2,cam.ViewportSize.Y/2)
        elseif ESPState.TracerOrigin=="Mouse" then
            local ok,mouseLoc=pcall(function() return game:GetService("UserInputService"):GetMouseLocation() end)
            if ok and mouseLoc then origin=mouseLoc end
        end
        obj.Tracer.From=origin
        obj.Tracer.To=Vector2.new(centerX,bot)
        obj.Tracer.Color=tracerClr
    end
    obj.Name.Visible=ESPState.Name
    if ESPState.Name then
        local p=ESP_PLAYER_MAP[char] or PlayerService:GetPlayerFromCharacter(char)
        local rawName=p and (p.DisplayName or p.Name) or char.Name
        obj.Name.Text=rawName
        obj.Name.Size=ESPState.TextSize
        obj.Name.Position=Vector2.new(centerX,top-(ESPState.TextSize+2))
        obj.Name.Color=nameClr
        obj.Name.Outline=ESPState.TextOutline
    end
    obj.DistText.Visible=ESPState.Distance
    if ESPState.Distance then
        obj.DistText.Text=math.floor(dist).."m"
        obj.DistText.Size=math.max(ESPState.TextSize-2,8)
        obj.DistText.Position=Vector2.new(centerX,bot+3)
        obj.DistText.Color=distClr
        obj.DistText.Outline=ESPState.TextOutline
    end
    local headPart=char:FindFirstChild("Head")
    obj.HeadDot.Visible=ESPState.HeadDot and headPart~=nil
    if obj.HeadDot.Visible then
        local headPos,onS=worldToScreen(headPart.Position)
        obj.HeadDot.Position=headPos
        obj.HeadDot.Radius=math.clamp(width*0.12,2,6)

                if UserGameSettings.TouchCameraMovementSensitivity ~= SensitivityLevel then
                    UserGameSettings.TouchCameraMovementSensitivity = SensitivityLevel
                end
            elseif UserGameSettings.TouchCameraMovementSensitivity ~= TouchSensitivity then
                UserGameSettings.TouchCameraMovementSensitivity = TouchSensitivity
            end
        end
    end
    local function LocalFunction3()
        if OptionsConfig.MobileSensToggle.Value then
            if not (NilValue and NilValue.Connected) then
                NilValue = RunService.Heartbeat:Connect(LocalFunction2)
            end
        else
            if NilValue and NilValue.Connected then
                NilValue:Disconnect()
                NilValue = nil
            end
            LocalFunction2()
        end
    end
    MobileSensToggle:OnChanged(LocalFunction3)
    MobileSensSlider:OnChanged(LocalFunction2)
    game:BindToClose(function()
        if UserGameSettings then
            UserGameSettings.TouchCameraMovementSensitivity = TouchSensitivity
        end
    end)
    task.spawn(LocalFunction3)
end
TabConfig.Settings:CreateButton({
    Title = "Server Hop",
    Description = "Finds and teleports you to a new server.",
    Callback = function()
        local PlaceId = game.PlaceId
        local EmptyTable = {}
        local EmptyString = ""
        local CurrentHour = os.date("!*t").hour
        if not pcall(function()
            EmptyTable = HttpService:JSONDecode(readfile("NotSameServers.json"))
        end) then
            table.insert(EmptyTable, CurrentHour)
            local HttpService = HttpService
            writefile("NotSameServers.json", HttpService:JSONEncode(EmptyTable))
        end
        function teleportReturner()
            local DataContainer
            if EmptyString ~= "" then
                DataContainer = HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100&cursor=" .. EmptyString))
            else
                DataContainer = HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"))
            end
            if DataContainer.nextPageCursor and (DataContainer.nextPageCursor ~= "null" and DataContainer.nextPageCursor ~= nil) then
                EmptyString = DataContainer.nextPageCursor
            end
            local DataKey, DataValue, DataObject = pairs(DataContainer.data)
            local ZeroCount = 0
            while true do
                local IdContainer
                DataObject, IdContainer = DataKey(DataValue, DataObject)
                if DataObject == nil then
                    break
                end
                local BooleanValue = true
                local IdString = tostring(IdContainer.id)
                if tonumber(IdContainer.maxPlayers) > tonumber(IdContainer.playing) then
                    local TableKey, TableValue, TableObject = pairs(EmptyTable)
                    while true do
                        local TempVariable2
                        TableObject, TempVariable2 = TableKey(TableValue, TableObject)
                        if TableObject == nil then
                            break
                        end
                        if ZeroCount == 0 then
                            if tonumber(CurrentHour) ~= tonumber(TempVariable2) then
                                pcall(function()
                                    delfile("NotSameServers.json")
                                    EmptyTable = {}
                                    table.insert(EmptyTable, CurrentHour)
                                end)
                            end
                        elseif IdString == tostring(TempVariable2) then
                            BooleanValue = false
                        end
                        ZeroCount = ZeroCount + 1
                    end
                    if BooleanValue == true then
                        table.insert(EmptyTable, IdString)
                        task.wait()
                        pcall(function()
                            local HttpService1 = HttpService
                            writefile("NotSameServers.json", HttpService1:JSONEncode(EmptyTable))
                            task.wait()
                            TeleportService:TeleportToPlaceInstance(PlaceId, IdString, LocalPlayer)
                        end)
                        task.wait(4)
                    end
                end
            end
        end
        function teleport()
            while task.wait() do
                pcall(function()
                    teleportReturner()
                    if EmptyString ~= "" then
                        teleportReturner()
                    end
                end)
            end
        end
        teleport()
    end
})
